1. 声明投射关系时添加说明

```cpp
// EditAnywhere 任何地方
// BlueprintCallable 蓝图调用

UPROPERTY(EditAnywhere, Category="移动平台", meta=(DisplayName="移动距离偏差", ToolTip="到达端点时允许的误差"))
float MoveDistanceOffset = 0.0f;
// `DisplayName`：在蓝图/细节面板显示中文名
// `ToolTip`：鼠标悬停提示中文
UFUNCTION(BlueprintCallable, meta=(DisplayName="开始移动", ToolTip="启动平台往返移动"))
void StartMoving();

```

2. 使用别名
```cpp
void AMovingPlatform::Tick(float DeltaTime)
{
    auto& 起点位置 = StartLocation;             // 引用别名，只在 cpp 有效，UHT 不会管
    auto& 平台速度 = PlatformVelocity;
    float 移动距离 = FVector::Dist(起点位置, GetActorLocation());

    if (移动距离 >= MoveDistance)
    {
        FVector 移动方向 = 平台速度.GetSafeNormal();
        FVector 新起点位置 = 起点位置 + 移动方向 * MoveDistance;
        SetActorLocation(新起点位置);
        起点位置 = 新起点位置;
        平台速度 = -平台速度;
    }
}
```
